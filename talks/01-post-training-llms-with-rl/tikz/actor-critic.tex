% Actor-critic: two trained networks; the critic's prediction turns a return into an advantage.
\documentclass[border=8pt]{standalone}
\input{pt-style}
\begin{document}
\begin{tikzpicture}[font=\small, block/.append style={minimum height=1.0cm}]
  \node[tname] (s) at (0,0) {state $s_t$};
  \node[junction] (j) at (1.25,0) {};
  \draw[wire] (s) -- (j);
  \node[blocka, minimum width=3.1cm] (act) at (3.6,1.45) {\textbf{actor} $\pi_\theta(a\mid s)$\\[-1pt]{\footnotesize chooses the action}};
  \node[block, minimum width=3.1cm] (cri) at (3.6,-1.45) {\textbf{critic} $V_\psi(s)$\\[-1pt]{\footnotesize predicts the return from $s$}};
  \draw[flow] (j) |- (act.west);
  \draw[flow] (j) |- (cri.west);
  \node[block, fill=grey100, minimum width=2.6cm] (env) at (8.2,1.45) {\textbf{environment}\\[-1pt]{\footnotesize returns rewards}};
  \draw[flow] (act) -- node[above, dims] {$a_t$} (env);
  \node[block, draw=ink, line width=1.0pt, minimum width=2.6cm] (adv) at (8.2,-1.45) {\textbf{advantage}\\[-1pt]$\hat A_t = R_t - V_\psi(s_t)$};
  \draw[flow] (env) -- node[right, dims] {return $R_t$} (adv);
  \draw[flow] (cri) -- node[above, dims] {$V_\psi(s_t)$} (adv);
  % update path, routed around the outside
  \draw[flowa] (adv.east) -- ++(0.55,0) |- ($(act.north)+(0,0.75)$) -- (act.north);
  \node[dims, anchor=south] at (6.4,2.95) {$\hat A_t$ weights the actor's update};
  \mka{($(act.north west)+(-0.1,0.12)$)}{1};
  \mk{($(cri.south west)+(-0.1,-0.12)$)}{2};
  \mk{($(adv.south east)+(0.12,-0.12)$)}{3};
  \notesat{(10.9,3.2)}{6.6cm}
  \notea{1}{\textbf{Actor}: the policy. Its update is the policy gradient, with each action weighted by $\hat A_t$.};
  \note{2}{\textbf{Critic}: a second network trained by regression to predict the return, $\min_\psi (V_\psi(s_t) - R_t)^2$.};
  \note{3}{\textbf{Advantage}: what happened minus what the critic expected. Positive means better than expected.};
  \symbols{$\theta$ & actor parameters \\ $\psi$ & critic parameters \\ $R_t$ & return from step $t$};
\end{tikzpicture}
\end{document}
