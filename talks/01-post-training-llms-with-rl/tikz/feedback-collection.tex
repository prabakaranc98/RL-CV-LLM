% How preference data is collected: sample K responses, a person ranks them, the ranking becomes K-choose-2 pairs.
\documentclass[border=8pt]{standalone}
\input{pt-style}
\begin{document}
\begin{tikzpicture}[font=\small,
    rsp/.style={block, minimum width=1.9cm, minimum height=0.5cm, font=\footnotesize, inner sep=2pt},
    rnk/.style={block, minimum width=1.6cm, minimum height=0.5cm, font=\footnotesize, inner sep=2pt},
    par/.style={block, minimum width=1.25cm, minimum height=0.5cm, font=\footnotesize, inner sep=2pt},
    grp/.style={draw=grey300, line width=0.6pt, rounded corners=5pt, inner sep=7pt},
    hdr/.style={font=\footnotesize\bfseries, text=grey500, anchor=south}]
  % 1: prompt and K sampled responses
  \node[block, minimum width=1.3cm, minimum height=0.9cm] (x) at (0,-1.95) {prompt $x$};
  \foreach \l [count=\k] in {A,B,C,D} \node[rsp] (s\k) at (2.95,{-0.9-0.7*(\k-1)}) {response \l};
  \coordinate (t) at (1.25,-1.95);
  \draw[wire] (x.east) -- (t); \node[junction] at (t) {};
  \draw[wire] (t |- s1.west) -- (t |- s4.west);
  \foreach \k in {1,...,4} \draw[flow] (t |- s\k.west) -- (s\k.west);
  \node[grp, fit=(s1)(s4)] (G1) {};
  % 2: a person ranks them
  \foreach \l/\r [count=\k] in {B/1st, A/2nd, D/3rd, C/4th} \node[rnk] (k\k) at (6.0,{-0.9-0.7*(\k-1)}) {\r: \l};
  \node[grp, fit=(k1)(k4)] (G2) {};
  \draw[flow] (G1.east) -- (G2.west);
  % 3: the ranking becomes pairs
  \foreach \pa/\pb [count=\k] in {B/A, B/D, B/C, A/D, A/C, D/C} {
    \pgfmathsetmacro{\cx}{ifthenelse(\k<4, 8.75, 10.15)}
    \pgfmathsetmacro{\cy}{-1.2-0.75*mod(\k-1,3)}
    \node[par] (p\k) at (\cx,\cy) {\pa\ $\succ$ \pb};}
  \node[grp, fit=(p1)(p6)] (G3) {};
  \draw[flow] (G2.east) -- (G3.west |- G2.east);
  % 4: training rows for the reward model
  \node[blocka, minimum width=2.3cm, minimum height=1.3cm, font=\footnotesize, align=center] (D) at (13.05,-1.95)
    {six training rows\\$(x,\ y_w,\ y_l)$};
  \draw[flowa] (G3.east |- D.west) -- (D.west);
  % headers (one row, aligned) with their markers
  \node[hdr] (h1) at ($(G1.north)+(0.15,0.12)$) {from the model};
  \node[hdr] (h2) at ($(G2.north |- G1.north)+(0.15,0.12)$) {a person ranks};
  \node[hdr] (h3) at ($(G3.north |- G1.north)+(0.15,0.12)$) {every pair};
  \node[hdr] (h4) at ($(D.north |- G1.north)+(0.15,0.12)$) {training data};
  \foreach \k in {1,2,3} \mk{($(h\k.west)+(-0.3,0.02)$)}{\k};
  \mka{($(h4.west)+(-0.3,0.02)$)}{4};
  % notes
  \notesat{(15.0,0.55)}{6.1cm}
  \note{1}{Sample $K$ responses to one prompt from the current model. InstructGPT showed labellers between 4 and 9.}
  \note{2}{A person orders them from best to worst. Comparing is easier for people than giving each answer an absolute score.}
  \note{3}{One ranking of $K$ responses gives $K(K-1)/2$ pairs: 6 for $K = 4$, 36 for $K = 9$.}
  \notea{4}{Each pair is one row: the preferred response $y_w$ and the rejected one $y_l$. The Bradley--Terry loss (next slide) trains the reward model on them.}
\end{tikzpicture}
\end{document}
